local function GetBestQuest()
    local success, levelVal = pcall(function()
        return LocalPlayer.Data.Level.Value
    end)
    local playerLevel = success and levelVal or 1
    local bestQuestID = nil
    local bestQuestData = nil
    local highestLevelFound = -1
    local searchAreas = {QuestModule}
    
    if QuestModule.RepeatableQuests then
        table.insert(searchAreas, QuestModule.RepeatableQuests)
    end
    
    for _, area in ipairs(searchAreas) do
        for id, data in pairs(area) do
            if type(data) == "table" and data.recommendedLevel then
                if playerLevel >= data.recommendedLevel and data.recommendedLevel > highestLevelFound then
                    highestLevelFound = data.recommendedLevel
                    bestQuestID = id
                    bestQuestData = data
                end
            end
        end
    end
    
    return bestQuestID, bestQuestData, playerLevel
end

local function GetAliveMob(mobName)
    local npcsFolder = Workspace:FindFirstChild("NPCs")
    if not npcsFolder then return nil end
    
    local targetMatchName = CleanString(mobName)
    local closestMob = nil
    local shortestDistance = math.huge
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    
    for _, mob in ipairs(npcsFolder:GetChildren()) do
        if mob:IsA("Model") then
            local humanoid = mob:FindFirstChild("Humanoid")
            local mobHrp = mob:FindFirstChild("HumanoidRootPart")
            
            if humanoid and mobHrp and humanoid.Health > 0 then
                local baseMobName = CleanString(string.gsub(mob.Name, "%d+$", ""))
                
                local displayName = humanoid.DisplayName or ""
                local cleanDisplayName = CleanString(string.gsub(displayName, "%s*%[.-%]$", ""))
                
                if baseMobName == targetMatchName or cleanDisplayName == targetMatchName then
                    if hrp then
                        local distance = (hrp.Position - mobHrp.Position).Magnitude
                        if distance < shortestDistance then
                            shortestDistance = distance
                            closestMob = mob
                        end
                    else
                        return mob
                    end
                end
            end
        end
    end
    
    return closestMob
end

local function GetClosestPortalIsland(targetPosition)
    local bestName     = nil
    local bestDistance = math.huge

    -- Portals live as children of island folders in workspace.
    -- Example: workspace.HuecoMundo.Portal_HuecoMundo
    -- We do a deep scan so we catch any nesting depth.
    local function ScanDescendants(parent)
        for _, child in ipairs(parent:GetChildren()) do
            if child.Name:sub(1, 7) == "Portal_" then
                local islandName = child.Name:sub(8) -- strip "Portal_" prefix
                local ok, cf = pcall(function() return child:GetPivot() end)
                if not ok or not cf then
                    -- fallback: try .CFrame if the instance is a BasePart
                    if child:IsA("BasePart") then
                        ok = true
                        cf = child.CFrame
                    end
                end
                if ok and cf then
                    local dist = (cf.Position - targetPosition).Magnitude
                    if dist < bestDistance then
                        bestDistance = dist
                        bestName     = islandName
                    end
                end
            end
            -- Recurse one level (island folder → portal)
            ScanDescendants(child)
        end
    end

    ScanDescendants(Workspace)
    return bestName -- nil if no portals found
end

local function GetMidPointCFrame(mobName)
    local npcsFolder = Workspace:FindFirstChild("NPCs")
    if not npcsFolder then return nil end

    local targetMatch = CleanString(mobName)

    -- ── Step 1: collect positions of all valid, alive, matching mobs ──────
    -- A mob is valid only if it has a Humanoid with health > 0 AND a
    -- HumanoidRootPart (used as the position anchor, consistent with every
    -- other part of the system).  Mobs without HRP are silently skipped so
    -- a single broken model can never corrupt the average.
    local centerSum = Vector3.new(0, 0, 0)
    local count     = 0

    for _, mob in ipairs(npcsFolder:GetChildren()) do
        if mob:IsA("Model") and mob.Parent then
            local humanoid = mob:FindFirstChild("Humanoid")
            local hrp      = mob:FindFirstChild("HumanoidRootPart")

            -- Guard: must be alive, must have a physical anchor
            if humanoid and hrp and humanoid.Health > 0 then
                local baseName = CleanString(string.gsub(mob.Name, "%d+$", ""))
                local dispName = CleanString(string.gsub(humanoid.DisplayName or "", "%s*%[.-%]$", ""))

                if baseName == targetMatch or dispName == targetMatch then
                    centerSum = centerSum + hrp.Position
                    count     = count + 1
                end
            end
        end
    end

    -- ── Step 2: require at least 2 mobs; single mob → caller uses normal teleport ──
    if count < 2 then return nil end

    -- ── Step 3: true mathematical center of the mob cluster ───────────────
    local midpoint = centerSum / count

    -- ── Step 4: apply Distance slider upward offset ───────────────────────
    -- The teleport position sits directly above the mob cluster centre.
    -- CFrame.lookAt(from, to) orients the character so it faces downward
    -- toward the midpoint, ensuring melee and ranged attacks connect with
    -- the mobs below rather than firing off in a random direction.
    local distance      = Settings.TpDistance
    local teleportPos   = midpoint + Vector3.new(0, distance, 0)

    return CFrame.lookAt(teleportPos, midpoint)
end

-- ─────────────────────────────────────────────────────────
-- ZIG ZAG STYLE — STABLE DIRECTIONAL REPOSITIONING
--
-- Design goals:
--   • Each chosen position is held for ~3.5 s so the player
--     stays in one spot long enough for hits to register and
--     the server to see sustained, plausible movement.
--   • Directions come from a named set so transitions feel
--     deliberate rather than random.
--   • A smooth-transition map prevents direct opposites and
--     immediate flip-flops (Left→Right, Behind→Front, etc).
--   • Distance jitter is ≤ ±0.5 studs so the player never
--     exceeds the configured Distance slider value by any
--     meaningful amount.
--   • Every tick the CFrame is RECOMPUTED from the mob's live
--     position, so if the mob moves the player follows it —
--     but the *direction* stays locked until the timer fires.
--
-- zigState fields (allocated per mob engagement):
--   .slot        – index into ZZ_DIRS currently locked in
--   .prevSlot    – slot used one rotation ago (prevents flip)
--   .lockedUntil – os.clock() timestamp when the lock expires
--   .jitter      – distance jitter frozen for this hold period
-- ─────────────────────────────────────────────────────────

-- Hold duration (seconds) between direction changes
local ZZ_HOLD_TIME = 3.5

-- Named directions: { rightMul, fwdMul, upMul }
-- Vectors are unit-normalised so distance scaling is exact.
local ZZ_DIRS = {
    --  idx   name              R       F      U
    {  0.00, -1.00,  0.00 },  --  1  Behind
    {  0.00,  1.00,  0.00 },  --  2  Front
    { -1.00,  0.00,  0.00 },  --  3  Left
    {  1.00,  0.00,  0.00 },  --  4  Right
    {  0.00,  0.00,  1.00 },  --  5  Above
    { -0.71, -0.71,  0.00 },  --  6  Diag: Left-Behind
    {  0.71, -0.71,  0.00 },  --  7  Diag: Right-Behind
    { -0.71,  0.71,  0.00 },  --  8  Diag: Left-Front
    {  0.71,  0.71,  0.00 },  --  9  Diag: Right-Front
    { -0.71,  0.00,  0.71 },  -- 10  Diag: Left-Above
    {  0.71,  0.00,  0.71 },  -- 11  Diag: Right-Above
}

-- Smooth-transition table: for each slot, a list of ALLOWED next slots.
-- Rule: no direct opposite, no slot that was just used (prevSlot is also
-- excluded at call time).  Diagonals are always safe transitions.
-- Direct-opposite pairs: 1↔2 (Behind↔Front), 3↔4 (Left↔Right), 6↔9, 7↔8
local ZZ_ALLOWED_NEXT = {
    [1]  = {3, 4, 5, 6, 7, 10, 11},      -- Behind  → not Front
    [2]  = {3, 4, 5, 8, 9, 10, 11},      -- Front   → not Behind
    [3]  = {1, 2, 5, 6, 8, 10},          -- Left    → not Right
    [4]  = {1, 2, 5, 7, 9, 11},          -- Right   → not Left
    [5]  = {1, 2, 3, 4, 6, 7, 8, 9},     -- Above   → any cardinal
    [6]  = {1, 2, 3, 5, 7, 10, 11},      -- DL-Back → not DR-Front
    [7]  = {1, 2, 4, 5, 6, 10, 11},      -- DR-Back → not DL-Front
    [8]  = {1, 2, 3, 5, 9, 10, 11},      -- DL-Fwd  → not DR-Back
    [9]  = {1, 2, 4, 5, 8, 10, 11},      -- DR-Fwd  → not DL-Back
    [10] = {1, 2, 3, 5, 6, 7, 8},        -- DL-Up   → not DR-Up
    [11] = {1, 2, 4, 5, 7, 9, 10},       -- DR-Up   → not DL-Up
}

local function _PickNextZZSlot(currentSlot, prevSlot)
    local candidates = ZZ_ALLOWED_NEXT[currentSlot] or {}
    -- Remove prevSlot from candidates to prevent flip-back
    local pool = {}
    for _, idx in ipairs(candidates) do
        if idx ~= prevSlot then
            table.insert(pool, idx)
        end
    end
    -- Safety fallback: if pool is empty (shouldn't happen), allow all except current
    if #pool == 0 then
        for i = 1, #ZZ_DIRS do
            if i ~= currentSlot then table.insert(pool, i) end
        end
    end
    return pool[math.random(1, #pool)]
end

-- Main function — call every tick.
-- Only mutates zigState when the hold timer has expired.
-- Returns a fresh CFrame recalculated from the mob's live position.
local function GetZigZagCFrame(targetHrp, zigState)
    local now = os.clock()

    -- Initialise on first call (slot 0 = uninitialised sentinel)
    if zigState.slot == 0 or now >= zigState.lockedUntil then
        local newSlot
        if zigState.slot == 0 then
            -- First position: start from Behind (slot 1) — safest opener
            newSlot = 1
        else
            newSlot = _PickNextZZSlot(zigState.slot, zigState.prevSlot)
        end
        zigState.prevSlot    = zigState.slot ~= 0 and zigState.slot or newSlot
        zigState.slot        = newSlot
        zigState.lockedUntil = now + ZZ_HOLD_TIME
        -- Freeze a tiny distance jitter for this hold period (≤ ±0.5 stud)
        zigState.jitter      = (math.random() - 0.5) * 1.0
    end

    local d   = math.max(1, Settings.TpDistance + zigState.jitter)
    local dir = ZZ_DIRS[zigState.slot]

    local right   = targetHrp.CFrame.RightVector
    local forward = targetHrp.CFrame.LookVector
    local worldUp = Vector3.new(0, 1, 0)

    local offset   = (right   * (dir[1] * d))
                   + (forward * (dir[2] * d))
                   + (worldUp * (dir[3] * d))

    -- Face toward the mob so melee/ranged hits register correctly
    return CFrame.new(targetHrp.Position + offset, targetHrp.Position)
end

local function GetTeleportCFrame(targetHrp)
    local targetPos = targetHrp.Position
    -- Use dungeon boss distance override if active, otherwise user's setting
    local distance  = (Settings.AutoDungeon and Settings.DungeonDistanceOverride) or Settings.TpDistance
    local offsetCFrame

    if Settings.RoundTeleport then
        -- Circular orbit: advance angle each call, orbit at RoundDistance radius
        local radius = Settings.RoundDistance
        local angle  = Settings._RoundAngle or 0
        -- Advance by RoundSpeed degrees per call
        angle = (angle + math.rad(Settings.RoundSpeed)) % (math.pi * 2)
        Settings._RoundAngle = angle

        local x = targetPos.X + radius * math.cos(angle)
        local z = targetPos.Z + radius * math.sin(angle)
        -- Keep player at same height as target to avoid falling
        local y = targetPos.Y
        offsetCFrame = CFrame.new(Vector3.new(x, y, z), targetPos)
    elseif Settings.TpStyle == "Above" then
        offsetCFrame = CFrame.new(targetPos + Vector3.new(0, distance, 0), targetPos)
    elseif Settings.TpStyle == "Below" then
        offsetCFrame = CFrame.new(targetPos + Vector3.new(0, -distance, 0), targetPos)
    elseif Settings.TpStyle == "Behind" then
        offsetCFrame = CFrame.new(targetPos + (targetHrp.CFrame.LookVector * -distance), targetPos)
    elseif Settings.TpStyle == "Front" then
        offsetCFrame = CFrame.new(targetPos + (targetHrp.CFrame.LookVector * distance), targetPos)
    end

    return offsetCFrame
end

local _tweenConn        = nil
local _tweenTarget      = nil
local _tweenFinalCF     = nil
local _tweenLastPos     = nil
local _tweenStuckFrames = 0
local _tweenStartTime   = 0
local _tweenMaxTime     = 0

local STUCK_MIN_MOVE   = 0.3
local STUCK_MAX_FRAMES = 3

local function _SnapToDestination(h, finalCF)
    local dest = finalCF.Position
    pcall(function()
        h.CFrame = CFrame.new(dest + Vector3.new(0, 5, 0), dest)
    end)
    task.delay(0.05, function()
        local char2 = LocalPlayer.Character
        local h2    = char2 and char2:FindFirstChild("HumanoidRootPart")
        if h2 then
            pcall(function() h2.CFrame = finalCF end)
        end
    end)
end

local function _CancelStepMove(hrp)
    if _tweenConn then
        _tweenConn:Disconnect()
        _tweenConn        = nil
        _tweenTarget      = nil
        _tweenFinalCF     = nil
        _tweenLastPos     = nil
        _tweenStuckFrames = 0
        _tweenStartTime   = 0
        _tweenMaxTime     = 0
    end
    if hrp then
        pcall(function()
            hrp.AssemblyLinearVelocity  = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end)
    end
end

local function TeleportTo(targetCFrame)
    local character = LocalPlayer.Character
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local targetPos = targetCFrame.Position
    local distance  = (hrp.Position - targetPos).Magnitude

    if Settings.TpMethod == "Instant" then
        _CancelStepMove(hrp)
        hrp.CFrame = targetCFrame
        hrp.AssemblyLinearVelocity = Vector3.zero
        return
    end

    if distance <= 8 then
        _CancelStepMove(hrp)
        hrp.CFrame = targetCFrame
        hrp.AssemblyLinearVelocity = Vector3.zero
        return
    end

    if _tweenConn and _tweenTarget and (_tweenTarget - targetPos).Magnitude < 2 then
        return
    end

    local speed     = math.max(1, Settings.TweenSpeed)
    local travelSec = (distance / speed) * 2.5

    _CancelStepMove(hrp)
    _tweenTarget      = targetPos
    _tweenFinalCF     = targetCFrame
    _tweenLastPos     = hrp.Position
    _tweenStuckFrames = 0
    _tweenStartTime   = os.clock()
    _tweenMaxTime     = math.max(travelSec, 1.5)

    _tweenConn = RunService.Heartbeat:Connect(function(dt)
        local char = LocalPlayer.Character
        local h    = char and char:FindFirstChild("HumanoidRootPart")
        if not h then
            _CancelStepMove(nil)
            return
        end

        local now       = os.clock()
        local toTarget  = _tweenTarget - h.Position
        local remaining = toTarget.Magnitude

        if remaining <= 4 then
            local cf = _tweenFinalCF
            _CancelStepMove(h)
            pcall(function() h.CFrame = cf end)
            return
        end

        if now - _tweenStartTime >= _tweenMaxTime then
            local cf = _tweenFinalCF
            _CancelStepMove(h)
            _SnapToDestination(h, cf)
            return
        end

        local moved = _tweenLastPos and (h.Position - _tweenLastPos).Magnitude or math.huge
        _tweenLastPos = h.Position

        if moved < STUCK_MIN_MOVE then
            _tweenStuckFrames = _tweenStuckFrames + 1
        else
            _tweenStuckFrames = 0
        end

        if _tweenStuckFrames >= STUCK_MAX_FRAMES then
            local cf = _tweenFinalCF
            _CancelStepMove(h)
            _SnapToDestination(h, cf)
            return
        end

        local dir    = toTarget.Unit
        local step   = math.min(speed * dt, remaining)
        local newPos = h.Position + dir * step

        pcall(function()
            h.CFrame = CFrame.lookAt(newPos, newPos + dir)
        end)
    end)
end

end

local function SanitizeName(name)
    if type(name) ~= "string" then return "" end
    name = name:gsub("%c", "")
    name = name:gsub("\xc2\xa0", "")
    name = name:gsub("^%s+", ""):gsub("%s+$", "")
    return name
end

local function HandleAutoEquip()
    if not Settings.AutoEquip then return end

    local character = LocalPlayer.Character
    if not character then return end

    local humanoid = character:FindFirstChild("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return end

    local backpack = LocalPlayer:FindFirstChild("Backpack")

    local selectedWeapon = SanitizeName(Settings.SelectedWeapon)

    -- ── No weapon selected: fallback to first available tool ─────────────
    -- If the player already has any tool equipped, nothing needs to happen.
    -- Otherwise equip the first Tool found in the Backpack so combat never
    -- stalls just because the dropdown was left blank.
    if selectedWeapon == "" then
        local currentTool = character:FindFirstChildOfClass("Tool")
        if currentTool then return end  -- already holding something — done

        if not backpack then return end
        local firstTool = backpack:FindFirstChildOfClass("Tool")
        if firstTool then
            humanoid:EquipTool(firstTool)
        end
        return
    end

    -- ── Weapon selected: original targeted-equip logic ────────────────────
    local currentTool = character:FindFirstChildOfClass("Tool")
    if currentTool and SanitizeName(currentTool.Name) == selectedWeapon then
        return
    end

    if currentTool then
        humanoid:UnequipTools()
    end

    if not backpack then return end

    local toolToEquip

    for _, item in ipairs(backpack:GetChildren()) do
        if item:IsA("Tool") and SanitizeName(item.Name) == selectedWeapon then
            toolToEquip = item
            break
        end
    end

    if not toolToEquip then
        for _, item in ipairs(character:GetChildren()) do
            if item:IsA("Tool") and SanitizeName(item.Name) == selectedWeapon then
                return
            end
        end
    end

    if toolToEquip then
        humanoid:EquipTool(toolToEquip)
    end
end

-- Re-equip weapon after respawn for AutoFarmLevel
-- Tools are returned to the Backpack after a short server delay on respawn.
-- CharacterAdded fires before the backpack is repopulated, so we wait for
-- the first tool to appear rather than using a fixed sleep.
LocalPlayer.CharacterAdded:Connect(function(char)
    if not Settings.AutoLevel then return end
    local hum = char:WaitForChild("Humanoid", 10)
    if not hum then return end
    local backpack = LocalPlayer:WaitForChild("Backpack", 10)
    if not backpack then return end
    -- Wait until at least one tool is available (up to 5s)
    local tool = backpack:FindFirstChildOfClass("Tool")
    if not tool then
        local elapsed = 0
        repeat
            task.wait(0.1)
            elapsed = elapsed + 0.1
            tool = backpack:FindFirstChildOfClass("Tool")
        until tool or elapsed >= 5
    end
    if tool and hum.Health > 0 then
        HandleAutoEquip()
    end
end)

local _visitedServers  = {}
local _lastHopAttempt  = 0
local _isHopping       = false

local function IsServerAvailable(s)
    return type(s.id) == "string"
        and s.id ~= ""
        and s.id ~= game.JobId
        and not _visitedServers[s.id]
        and type(s.playing) == "number"
        and type(s.maxPlayers) == "number"
        and s.maxPlayers > 0
        and s.playing < s.maxPlayers
end

local function FetchServers(placeId, minPlayers, maxPages)
    local servers = {}
    local cursor  = ""
    local pages   = 0
    local order   = minPlayers and "Asc" or "Desc"

    repeat
        local url = ("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=%s&limit=100%s")
            :format(placeId, order, cursor ~= "" and ("&cursor=" .. cursor) or "")

        local ok, raw = pcall(game.HttpGet, game, url)
        if not ok or not raw then break end

        local dok, data = pcall(HttpService.JSONDecode, HttpService, raw)
        if not dok or not data or not data.data then break end

        for _, s in ipairs(data.data) do
            if IsServerAvailable(s) then
                table.insert(servers, s)
            end
        end

        cursor = data.nextPageCursor or ""
        pages  = pages + 1
    until cursor == "" or pages >= maxPages

    table.sort(servers, function(a, b)
        if minPlayers then
            return a.playing < b.playing
        else
            return a.playing > b.playing
        end
    end)

    return servers
end

local function TryTeleport(placeId, serverId)
    local result   = nil
    local done     = false

    local conn = TeleportService.TeleportInitFailed:Connect(function(player, failReason)
        if player == LocalPlayer then
            result = failReason
            done   = true
        end
    end)

    local ok, err = pcall(function()
        TeleportService:TeleportToPlaceInstance(placeId, serverId, LocalPlayer)
    end)

    if not ok then
        pcall(conn.Disconnect, conn)
        return false, tostring(err)
    end

    local elapsed = 0
    while not done and elapsed < 12 do
        task.wait(0.2)
        elapsed = elapsed + 0.2
    end

    pcall(conn.Disconnect, conn)

    if done then
        return false, tostring(result)
    end

    return true, nil
end

local function ServerHop(minPlayers)
    if _isHopping then return end
    local now = tick()
    if now - _lastHopAttempt < 10 then return end
    _lastHopAttempt = now
    _isHopping = true

    task.spawn(function()
        local placeId = game.PlaceId
        local MAX_ATTEMPTS = 3

        for attempt = 1, MAX_ATTEMPTS do
            local servers = FetchServers(placeId, minPlayers, 5)

            if #servers == 0 then
                if attempt == MAX_ATTEMPTS then
                    _visitedServers = {}
                end
                task.wait(3)
                continue
            end

            local hopped = false

            for _, target in ipairs(servers) do
                if not IsServerAvailable(target) then continue end

                _visitedServers[target.id] = true

                local success, reason = TryTeleport(placeId, target.id)

                if success then
                    hopped = true
                    break
                end
            end

            if hopped then break end

            if attempt == MAX_ATTEMPTS then
                _visitedServers = {}
            end

            task.wait(2)
        end

        _isHopping = false
    end)
end

local _statDebounce = false

local function RunAutoStats()
    if _statDebounce then return end
    if not Settings.AutoStats then return end

    local dataFolder = LocalPlayer:FindFirstChild("Data")
    if not dataFolder then return end

    local pointsVal = dataFolder:FindFirstChild("StatPoints")
    if not pointsVal or pointsVal.Value <= 0 then return end

    local points = pointsVal.Value

    local liveStats = {}
    if Settings.StatListenCaps then
        pcall(function()
            local result = GetPlayerStatsRemote:InvokeServer()
            if result and result.Stats then
                liveStats = result.Stats
            end
        end)
    end

    local sortedStats = {}
    for name, data in pairs(Settings.StatPriorities) do
        if data.Selected then
            local cap     = Settings.StatListenCaps and (Settings.StatCaps[name] or 0) or 0
            local current = liveStats[name] or 0
            if cap <= 0 or current < cap then
                table.insert(sortedStats, { Name = name, Priority = data.Priority, Current = current })
            end
        end
    end

    if #sortedStats == 0 then return end

    table.sort(sortedStats, function(a, b) return a.Priority < b.Priority end)

    _statDebounce = true

    local spent = {}

    if Settings.StatUpgradeType == "Priority" then
        local top     = sortedStats[1]
        local cap     = Settings.StatListenCaps and (Settings.StatCaps[top.Name] or 0) or 0
        local alloc   = (cap > 0) and math.min(points, cap - top.Current) or points
        if alloc > 0 then
            pcall(function() AllocateStatRemote:FireServer(top.Name, alloc) end)
            spent[top.Name] = alloc
        end

    elseif Settings.StatUpgradeType == "Equal" then
        local remaining = points
        local base      = math.floor(remaining / #sortedStats)
        local leftover  = remaining - base * #sortedStats

        for i, statInfo in ipairs(sortedStats) do
            local cap   = Settings.StatListenCaps and (Settings.StatCaps[statInfo.Name] or 0) or 0
            local share = base + (i <= leftover and 1 or 0)
            local alloc = (cap > 0) and math.min(share, cap - statInfo.Current) or share
            if alloc > 0 then
                pcall(function() AllocateStatRemote:FireServer(statInfo.Name, alloc) end)
                spent[statInfo.Name] = alloc
            end
        end
    end

    if Settings.StatNotifications then
        local parts = {}
        for sName, sAmt in pairs(spent) do
            table.insert(parts, sName .. " +" .. sAmt)
        end
        if #parts > 0 and Library then
            Library:Notification({
                Title       = "Auto Stats",
                Description = table.concat(parts, "  |  "),
                Duration    = 3,
            })
        end
    end

    task.delay(Settings.StatDelay, function()
        _statDebounce = false
    end)
end

--   ≥ 70 %  →  green   #44DD44
--   ≥ 40 %  →  yellow  #FFE044
--   ≥ 15 %  →  orange  #FF8C00
--   <  15 % →  red     #FF4444
-- "Off" task renders in soft red to signal the system is inactive.
-- ─────────────────────────────────────────────────────────
local function BuildFarmStatus(taskLabel, mobName, curHp, maxHp, progress)
    local LBL = "#999999"   -- dimmed label colour
    local VAL = "#FFFFFF"   -- bright value colour
    local OFF = "#CC4444"   -- soft red for disabled state

    -- Task line
    local taskColor = (taskLabel == "Off") and OFF or VAL
    local line1 = "<font color='"..LBL.."'><b>Current Task</b></font>  "
               .. "<font color='"..taskColor.."'>"..tostring(taskLabel).."</font>"

    -- Target Mob line
    local mobVal = (mobName and mobName ~= "") and tostring(mobName) or "—"
    local line2 = "<font color='"..LBL.."'><b>Target Mob</b></font>    "
               .. "<font color='"..VAL.."'>"..mobVal.."</font>"

    -- Mob Health line
    local healthStr
    if curHp and maxHp and maxHp > 0 then
        local pct    = math.clamp(curHp / maxHp * 100, 0, 100)
        local cur    = math.floor(curHp + 0.5)
        local max    = math.floor(maxHp + 0.5)
        local pctFmt = string.format("%.0f%%", pct)
        local hpColor
        if pct >= 70 then
            hpColor = "#44DD44"
        elseif pct >= 40 then
            hpColor = "#FFE044"
        elseif pct >= 15 then
            hpColor = "#FF8C00"
        else
            hpColor = "#FF4444"
        end
        healthStr = "<font color='"..VAL.."'>"..cur.." / "..max.."  </font>"
                 .. "<font color='"..hpColor.."'>(" .. pctFmt .. ")</font>"
    else
        healthStr = "<font color='"..LBL.."'>N/A</font>"
    end
    local line3 = "<font color='"..LBL.."'><b>Mob Health</b></font>    "..healthStr

    -- Quest Progress line
    local progVal = (progress and progress ~= "") and tostring(progress) or "—"
    local line4 = "<font color='"..LBL.."'><b>Quest Progress</b></font> "
               .. "<font color='"..VAL.."'>"..progVal.."</font>"

    return line1.."\n"..line2.."\n"..line3.."\n"..line4
end

-- Build mob list from QuestModule npcType entries (same source as AutoLevel)
local function BuildMobList()
    local seen = {}
    local list = {}
    local searchAreas = {QuestModule}
    if QuestModule.RepeatableQuests then
        table.insert(searchAreas, QuestModule.RepeatableQuests)
    end
    for _, area in ipairs(searchAreas) do
        for _, data in pairs(area) do
            if type(data) == "table" and data.requirements then
                for _, req in ipairs(data.requirements) do
                    local npc = req.npcType
                    if npc and not seen[npc] then
                        seen[npc] = true
                        table.insert(list, npc)
                    end
                end
            end
        end
    end
    table.sort(list)
    return #list > 0 and list or {"No mobs found"}
end

    end

    local IDLE_TEXT = "<font color='#555555'><i>No dungeon active.</i></font>"

        local timer   = timerLabel.Text
        local enemies = killsLabel.Text

            .. "</font>\n<font color='#FFFFFF'><b>Enemies Alive:</b></font> <font color='#DDDDDD'>" .. enemies .. "</font>"

        timerLabel:GetPropertyChangedSignal("Text"):Connect(function()

        end),
        killsLabel:GetPropertyChangedSignal("Text"):Connect(function()
            DungeonStatusParagraph:SetText(BuildStatusText())

local function BossStatusText(state, targetName, hp, maxHp, nextTarget, distance, positionOn, timeEngaged)
    local LINE = "\n"
    local DIM  = "<font color='#282828'></font>"

    if not state then
        return  "<font color='#888888' size='13'>AUTO BOSS</font><font color='#1e1e1e'>  ────────────────────</font>" .. LINE ..
                "<font color='#2a2a2a'>──────────────────────────</font>" .. LINE ..
                "<font color='#333333'>  Status</font>   <font color='#222222'>│</font>  <font color='#3a3a3a'><i>Offline</i></font>"
    end

    -- HP color thresholds
    local hpColor = "#4caf7d"
    local hpPct   = 0
    if maxHp and maxHp > 0 then
        hpPct = math.floor((hp / maxHp) * 100)
        if hpPct < 35 then
            hpColor = "#e05252"
        elseif hpPct < 65 then
            hpColor = "#e8a230"
        end
    end

    -- HP string: simple value / max with percentage
    local hpStr
    if maxHp and maxHp > 0 then
        hpStr =
            "<font color='" .. hpColor .. "'><b>" .. math.floor(hp) .. "</b></font>" ..
            "<font color='#2e2e2e'> / </font>" ..
            "<font color='#555555'>" .. math.floor(maxHp) .. "</font>" ..
            "  <font color='" .. hpColor .. "'>" .. hpPct .. "%</font>"
    else
        hpStr = "<font color='#333333'>—</font>"
    end

    -- Time format
    local timeStr = "—"
    if timeEngaged then
        if timeEngaged >= 3600 then
            timeStr = string.format("%dh %dm", math.floor(timeEngaged / 3600), math.floor((timeEngaged % 3600) / 60))
        elseif timeEngaged >= 60 then
            timeStr = string.format("%dm %ds", math.floor(timeEngaged / 60), timeEngaged % 60)
        else
            timeStr = string.format("%ds", timeEngaged)
        end
    end

    local distStr  = distance and string.format("%.0f studs", distance) or "—"

    local posStr   = positionOn
        and "<font color='#4caf7d'>On</font>"
        or  "<font color='#444444'>Off</font>"

    local targetStr = targetName
        and ("<font color='#6aa9f5'><b>" .. targetName .. "</b></font>")
        or  "<font color='#383838'><i>Scanning...</i></font>"

    local nextStr = (nextTarget and nextTarget ~= targetName)
        and ("<font color='#707070'>" .. nextTarget .. "</font>")
        or  "<font color='#2e2e2e'>—</font>"

    local col1 = "<font color='#404040'>"
    local col2 = "</font>  <font color='#1e1e1e'>│</font>  "

    return
        "<font color='#aaaaaa' size='13'><b>AUTO BOSS</b></font>  <font color='#1e1e1e'>────────────────</font>" .. LINE ..
        "<font color='#00e676' size='12'><b>Active</b></font>" .. LINE ..
        "<font color='#1c1c1c'>──────────────────────────</font>" .. LINE ..
        col1 .. "Target"   .. col2 .. targetStr .. LINE ..
        col1 .. "Health"   .. col2 .. hpStr .. LINE ..
        "<font color='#1c1c1c'>──────────────────────────</font>" .. LINE ..
        col1 .. "Next"     .. col2 .. nextStr .. LINE ..
        col1 .. "Distance" .. col2 .. "<font color='#555555'>" .. distStr .. "</font>" .. LINE ..
        col1 .. "Position" .. col2 .. posStr .. LINE ..
        col1 .. "Engaged"  .. col2 .. "<font color='#555555'>" .. timeStr .. "</font>"
end

-- ─── Alive Bosses Tracker (event-driven) ────────────────
local AliveBossesParagraph = BossStatusSection:Paragraph({
    Name = "Alive Bosses",
    Text = "• None",
    Icon = "list"
})

do
    local BOSS_TAG      = "  <font color='#8B0000'>[ Boss ]</font>"
    local NONE_TEXT     = "<font color='#555555'>•  None</font>"

    -- rawName -> displayName reverse lookup (built once)
    local rawToDisplay = {}
    for _, displayName in ipairs(BossDisplayNames) do
        local rawNames = BossNameMap[displayName]
        if rawNames then
            for _, rawName in ipairs(rawNames) do
                rawToDisplay[rawName] = displayName
            end
        end
    end

    -- alive set: displayName -> true (one entry per display name)
    local aliveBosses = {}
    -- track per-model connections to detect death via Humanoid.Died
    local bossConnections = {} -- rawName -> connection

    local debounceThread = nil
    local function ScheduleUIUpdate()
        if debounceThread then
            task.cancel(debounceThread)
        end
        debounceThread = task.delay(0.15, function()
            debounceThread = nil
            local lines = {}
            for _, displayName in ipairs(BossDisplayNames) do
                if aliveBosses[displayName] then
                    table.insert(lines,
                        "<font color='#cccccc'>•  " .. displayName .. "</font>" .. BOSS_TAG)
                end
            end
            AliveBossesParagraph:SetText(#lines > 0 and table.concat(lines, "\n") or NONE_TEXT)
        end)
    end

    local function TrackBoss(model)
        local rawName    = model.Name
        local displayName = rawToDisplay[rawName]
        if not displayName then return end

        local hum = model:FindFirstChild("Humanoid")
        if not hum or hum.Health <= 0 then return end

        -- Mark alive (only if not already tracked for this display name)
        aliveBosses[displayName] = true
        ScheduleUIUpdate()

        -- Disconnect any previous connection for this raw name
        if bossConnections[rawName] then
            bossConnections[rawName]:Disconnect()
        end

        bossConnections[rawName] = hum.Died:Connect(function()
            if bossConnections[rawName] then
                bossConnections[rawName]:Disconnect()
                bossConnections[rawName] = nil
            end
            -- Only clear if no other variant of this display name is still alive
            local npcsFolder = Workspace:FindFirstChild("NPCs")
            local stillAlive = false
            if npcsFolder then
                for _, rn in ipairs(BossNameMap[displayName] or {}) do
                    local m = npcsFolder:FindFirstChild(rn)
                    if m then
                        local h = m:FindFirstChild("Humanoid")
                        if h and h.Health > 0 then
                            stillAlive = true
                            break
                        end
                    end
                end
            end
            if not stillAlive then
                aliveBosses[displayName] = nil
            end
            ScheduleUIUpdate()
        end)
    end

    -- Scan existing NPCs folder on load
    task.spawn(function()
        local npcsFolder = Workspace:WaitForChild("NPCs", 10)
        if not npcsFolder then return end

        for _, model in ipairs(npcsFolder:GetChildren()) do
            if model:IsA("Model") and rawToDisplay[model.Name] then
                TrackBoss(model)
            end
        end

        -- Watch for new spawns
        npcsFolder.ChildAdded:Connect(function(child)
            if child:IsA("Model") and rawToDisplay[child.Name] then
                -- Small wait for Humanoid to fully initialize
                task.delay(0.1, function()
                    TrackBoss(child)
                end)
            end
        end)

        -- Watch for instant removals (e.g. despawn without Died firing)
        npcsFolder.ChildRemoved:Connect(function(child)
            local displayName = rawToDisplay[child.Name]
            if not displayName then return end
            if bossConnections[child.Name] then
                bossConnections[child.Name]:Disconnect()
                bossConnections[child.Name] = nil
            end
            -- Re-check if another variant is still alive
            local stillAlive = false
            for _, rn in ipairs(BossNameMap[displayName] or {}) do
                local m = npcsFolder:FindFirstChild(rn)
                if m then
                    local h = m:FindFirstChild("Humanoid")
                    if h and h.Health > 0 then
                        stillAlive = true
                        break
                    end
                end
            end
            if not stillAlive then
                aliveBosses[displayName] = nil
            end
            ScheduleUIUpdate()
        end)
    end)
end

local BossSummonSection = BossTab:Section({Name = "Summon", Side = 1, Icon = "flame"})

local SummonState = {
    SelectedBoss       = "",
    SelectedDifficulty = "Normal",
    AutoSummon         = false,
    CheckInterval      = 30,
}

local summonLoopThread = nil

-- Remote reference (resolved once)
local RequestSpawnRemote = ReplicatedStorage
    :WaitForChild("Remotes")
    :WaitForChild("RequestSpawnStrongestBoss")

-- Workspace name pattern per boss type + difficulty
local SummonBossPatterns = {
    ["Strongest Today"]   = "StrongestofTodayBoss_",
    ["Strongest History"] = "StrongestofHistoryBoss_",
}

-- Remote key per boss type
local SummonRemoteKeys = {
    ["Strongest Today"]   = "StrongestToday",
    ["Strongest History"] = "StrongestHistory",
}

local function BossExistsInWorkspace(bossType, difficulty)
    local npcsFolder = Workspace:FindFirstChild("NPCs")
    if not npcsFolder then return false end
    local pattern = SummonBossPatterns[bossType]
    if not pattern then return false end
    local targetName = pattern .. difficulty
    return npcsFolder:FindFirstChild(targetName) ~= nil
end

local function FireSummonRemote(bossType, difficulty)
    local key = SummonRemoteKeys[bossType]
    if not key then return end
    local ok, err = pcall(function()
        RequestSpawnRemote:FireServer(key, difficulty)
    end)
    if not ok then
        warn("[BossSummon] FireServer failed: " .. tostring(err))
    end
end

local function TrySummonBoss()
    if SummonState.SelectedBoss == "" then
        Library:Notification({ Title = "Boss Summon", Description = "No boss type selected.", Duration = 4 })
        return
    end

    if BossExistsInWorkspace(SummonState.SelectedBoss, SummonState.SelectedDifficulty) then
        Library:Notification({
            Title       = "Boss Summon",
            Description = SummonState.SelectedBoss .. " (" .. SummonState.SelectedDifficulty .. ") is already active.",
            Duration    = 4
        })
        return
    end

    FireSummonRemote(SummonState.SelectedBoss, SummonState.SelectedDifficulty)
    Library:Notification({
        Title       = "Boss Summon",
        Description = "Summoned " .. SummonState.SelectedBoss .. " on " .. SummonState.SelectedDifficulty .. ".",
        Duration    = 4
    })
end

local function StopSummonLoop()
    if summonLoopThread then
        task.cancel(summonLoopThread)
        summonLoopThread = nil
    end
end

local function UpdateStatPointsParagraph()
    local ok, err = pcall(function()
        local dataFolder = LocalPlayer:FindFirstChild("Data")
        if not dataFolder then
            StatPointsParagraph:SetText("<font color='#888888'>Waiting for data...</font>")
            return
        end

        local level = 0
        local levelVal = dataFolder:FindFirstChild("Level")
        if levelVal then level = levelVal.Value end

        local available = 0
        local pointsVal = dataFolder:FindFirstChild("StatPoints")
        if pointsVal then available = pointsVal.Value end

        local statsResult = nil
        pcall(function()
            local result = GetPlayerStatsRemote:InvokeServer()
            if result and result.Stats then
                statsResult = result.Stats
            end
        end)

        local statNames = {"Sword", "Power", "Melee", "Defense"}
        local lines = {
            string.format("<font color='#555555'>Level</font>   <font color='#7eb8ff'><b>%d</b></font>", level),
            string.format("<font color='#555555'>Points</font>  <font color='#00e676'><b>%d</b></font>", available),
            "",
        }

        for _, name in ipairs(statNames) do
            local current = (statsResult and statsResult[name]) or 0
            local cap     = Settings.StatListenCaps and (Settings.StatCaps[name] or 0) or 0
            local capStr  = cap > 0 and ("<font color='#f5a623'> / " .. cap .. "</font>") or ""
            table.insert(lines, string.format(
                "<font color='#888888'>%s</font>  <font color='#cccccc'><b>%d</b></font>%s",
                name, current, capStr
            ))
        end

        StatPointsParagraph:SetText(table.concat(lines, "\n"))
    end)

    if not ok then
        pcall(function()
            StatPointsParagraph:SetText("<font color='#888888'>Data unavailable.</font>")

local function GetSelectedBosses()
    local rawSelected = Library.Flags.SelectedBosses or {}
    local selected = {}
    for k, v in pairs(rawSelected) do
        if type(k) == "string" and v == true then
            selected[k] = true
        elseif type(k) == "number" and type(v) == "string" then
            selected[v] = true
        end
    end
    return selected
end

local function ScanBosses(selectedDisplay)
    local npcsFolder = Workspace:FindFirstChild("NPCs")
    if not npcsFolder then return {} end

    local pChar = LocalPlayer.Character
    local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")

    local found = {}

    for _, model in ipairs(npcsFolder:GetChildren()) do
        if not model:IsA("Model") then continue end

        local isBoss = model.Name:find("Boss")
        if not isBoss then
            local bv = model:FindFirstChild("Boss")
            isBoss = bv ~= nil and bv.Value == true
        end
        if not isBoss then continue end

        local hum = model:FindFirstChild("Humanoid")
        local hrp = model:FindFirstChild("HumanoidRootPart")
        if not hum or hum.Health <= 0 then continue end

        local displayName = BossRawToDisplay[model.Name]
        if not displayName then
            displayName = model.Name:gsub("Boss.*$", "")
            if displayName == "" then continue end
        end

        if not selectedDisplay[displayName] then continue end

        local dist = (pHrp and hrp) and (pHrp.Position - hrp.Position).Magnitude or math.huge
        table.insert(found, { displayName = displayName, model = model, distance = dist })
    end

    table.sort(found, function(a, b) return a.distance < b.distance end)
    return found
end

        if not ok or not targetCFrame then
            Library:Notification({ Title = "NPC Teleport", Description = "Could not resolve NPC position.", Duration = 4 })
            return
        end

        -- Offset slightly above model centre so player lands on top of it
        targetCFrame = targetCFrame * CFrame.new(0, 3, 4)

        if npcTeleportMethod == "Instant" then
            if npcActiveTween then
                npcActiveTween:Cancel()
                npcActiveTween = nil
            end
            hrp.CFrame = targetCFrame
            Library:Notification({ Title = "NPC Teleport", Description = "Teleported to " .. selectedNpc, Duration = 3 })

        elseif npcTeleportMethod == "Tween" then
            -- Cancel any active tween before starting a new one
            if npcActiveTween then
                npcActiveTween:Cancel()
                npcActiveTween = nil
            end

            local distance = (hrp.Position - targetCFrame.Position).Magnitude
            local duration = math.max(0.1, distance / npcTeleportTweenSpeed)

            npcActiveTween = TweenService:Create(
                hrp,
                TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out),
                { CFrame = targetCFrame }
            )
            npcActiveTween:Play()
            npcActiveTween.Completed:Connect(function()
                npcActiveTween = nil
            end)

            Library:Notification({
                Title       = "NPC Teleport",
                Description = "Tweening to " .. selectedNpc .. " (" .. string.format("%.0f", duration) .. "s)",
                Duration    = math.ceil(duration) + 1
            })
        end
    end
})

-- JSON encode: use game's HttpService for encoding only (not for sending)
local JsonService = pcall(function() return game:GetService("HttpService") end) and game:GetService("HttpService") or nil

local function JsonEncode(tbl)
    if JsonService then
        local ok, result = pcall(function() return JsonService:JSONEncode(tbl) end)
        if ok then return result end
    end
    -- Minimal manual fallback for simple flat tables
    local parts = {}
    for k, v in pairs(tbl) do
        local key = '"' .. tostring(k) .. '"'
        local val
        if type(v) == "string" then
            val = '"' .. v:gsub('"', '\\"') .. '"'
        elseif type(v) == "boolean" then
            val = tostring(v)
        elseif type(v) == "number" then
            val = tostring(v)
        else
            val = '"[complex]"'
        end
        table.insert(parts, key .. ":" .. val)
    end
    return "{" .. table.concat(parts, ",") .. "}"
end

-- Core send function
local function SendWebhook(eventType, detailsString)
    if not WH.Active then return end
    if not WH.Url or WH.Url == "" then
        warn("[Webhook] No URL configured.")
        return
    end
    if not whRequest then
        warn("[Webhook] No HTTP request function available in this executor.")
        return
    end

    local content = ""
    if WH.PingEveryone then
        content = "@everyone"
    elseif WH.PingHere then
        content = "@here"
    end

    local playerName = LocalPlayer.Name
    local timestamp  = os.date("!%d/%m/%Y %H:%M")

    -- ── Build per-event sections matching the image style ──────────────────────
    --
    -- Layout mirrors the reference:
    --   Header block  →  User / Status / System + --- divider
    --   Section 1     →  emoji  Bold Title  +  ```[+]/[*]``` code block
    --   Section 2     →  emoji  Bold Title  +  ├/│/└ tree lines (for loot style)
    --
    local section1, section2

    if eventType == "Level Up" then
        local oldLv, newLv = detailsString:match("Level (%d+).+(%d+)")
        oldLv = oldLv or "?"
        newLv = newLv or "?"
        local gain = (tonumber(newLv) or 0) - (tonumber(oldLv) or 0)

        -- Section 1: stats in [+]/[*] code block
        section1 = "\xF0\x9F\x93\x88 **Progress Report**\n" ..   -- 📈
            "```\n" ..
            "[+] Event  : Level Up\n" ..
            "[*] Before : Lv. " .. oldLv .. "\n" ..
            "[+] After  : Lv. " .. newLv .. "\n" ..
            "[*] Gain   : +" .. gain .. " Level\n" ..
            "```"

        -- Section 2: loot-tree style for context
        section2 = "\xF0\x9F\xAA\x99 **Session Info**\n" ..      -- 🪙
            "\u{251C} **Player** " .. playerName .. "\n" ..
            "\u{2502}  \u{2514} *Status:* Online\n" ..
            "\u{2514} *End of Log*"

    elseif eventType == "Gems Received" then
        local amount = tonumber(detailsString:match("%+?(%d+)")) or 0

        section1 = "\xF0\x9F\x92\x8E **Loot Overview**\n" ..     -- 💎
            "\u{251C} **Gems** x" .. amount .. "\n" ..
            "\u{2502}  \u{2514} *Type:* Currency\n" ..
            "\u{2514} *End of Log*"

        section2 = "\xF0\x9F\x93\x8A **Gain Summary**\n" ..      -- 📊
            "```\n" ..
            "[+] Type   : Gems\n" ..
            "[+] Amount : +" .. amount .. "\n" ..
            "[*] Player : " .. playerName .. "\n" ..
            "```"

    elseif eventType == "Money Received" then
        local amount = tonumber(detailsString:match("%+?(%d+)")) or 0

        section1 = "\xF0\x9F\xAA\x99 **Loot Overview**\n" ..     -- 🪙
            "\u{251C} **Beli** x" .. amount .. "\n" ..
            "\u{2502}  \u{2514} *Type:* Currency\n" ..
            "\u{2514} *End of Log*"

        section2 = "\xF0\x9F\x93\x8A **Gain Summary**\n" ..      -- 📊
            "```\n" ..
            "[+] Type   : Beli\n" ..
            "[+] Amount : +" .. amount .. "\n" ..
            "[*] Player : " .. playerName .. "\n" ..
            "```"

    else
        section1 = "\xF0\x9F\x94\x94 **System Event**\n" ..      -- 🔔
            "```\n[+] " .. tostring(detailsString) .. "\n```"
        section2 = nil
    end

    -- ── Assemble full description ───────────────────────────────────────────────
    local lines = {
        "**User:** " .. playerName,
        "**Status:** `Online` | **System:** `Imp Hub X`",
        "\u{2014}\u{2014}\u{2014}",
        "",
        section1,
    }
    if section2 then
        table.insert(lines, "")
        table.insert(lines, section2)
    end
    local description = table.concat(lines, "\n")

    -- ── Payload ────────────────────────────────────────────────────────────────
    local payload = {
        content = content,
        embeds = {{
            author      = { name = "Sailor Piece" },
            title       = "\xF0\x9F\x8F\x86 " .. eventType,   -- 🏆 <EventType>
            description = description,
            color       = 5763719,   -- Discord green #57F287
            footer      = {
                text = "Imp Hub X \u{2022} Session Logger | " .. timestamp .. " UTC"
            }
        }}
    }

    local body = JsonEncode(payload)

    local ok, err = pcall(function()
        whRequest({
            Url     = WH.Url,
            Method  = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body    = body,
        })
    end)

    if not ok then
        warn("[Webhook] Failed to send: " .. tostring(err))
    end
end

-- Debounce accumulators for Gems and Money
local gemDebounce  = { amount = 0, timer = nil }
local moneyDebounce = { amount = 0, timer = nil }

local function FlushGems()
    if gemDebounce.amount > 0 then
        SendWebhook("Gems Received", "+" .. gemDebounce.amount .. " Gems")
    end
    gemDebounce.amount = 0
    gemDebounce.timer  = nil
end

local function FlushMoney()
    if moneyDebounce.amount > 0 then
        SendWebhook("Money Received", "+" .. moneyDebounce.amount .. " Beli")
    end
    moneyDebounce.amount = 0
    moneyDebounce.timer  = nil
end

-- Wait for Data to exist (it may not be available immediately on join)
task.spawn(function()
    local data = LocalPlayer:WaitForChild("Data", 30)
    if not data then
        warn("[Webhook] LocalPlayer.Data not found within 30s.")
        return
    end

    -- Level Up
    local levelVal = data:WaitForChild("Level", 10)
    if levelVal then
        local previousLevel = levelVal.Value
        levelVal.Changed:Connect(function(newValue)
            if not WH.NotifyLevel then return end
            if newValue > previousLevel then
                local old = previousLevel
                previousLevel = newValue
                task.spawn(function()
                    SendWebhook("Level Up", "Level " .. old .. " → " .. newValue)
                end)
            else
                previousLevel = newValue
            end
        end)
    end

    -- Gems Increase (debounced 5s window)
    local gemsVal = data:WaitForChild("Gems", 10)
    if gemsVal then
        local prevGems = gemsVal.Value
        gemsVal.Changed:Connect(function(newValue)
            if not WH.NotifyGems then return end
            local delta = newValue - prevGems
            prevGems = newValue
            if delta <= 0 then return end
            gemDebounce.amount = gemDebounce.amount + delta
            if gemDebounce.timer then
                task.cancel(gemDebounce.timer)
            end
            gemDebounce.timer = task.delay(5, FlushGems)
        end)
    end

    -- Money Increase (debounced 5s window)
    local moneyVal = data:WaitForChild("Money", 10)
    if moneyVal then
        local prevMoney = moneyVal.Value
        moneyVal.Changed:Connect(function(newValue)
            if not WH.NotifyMoney then return end
            local delta = newValue - prevMoney
            prevMoney = newValue
            if delta <= 0 then return end
            moneyDebounce.amount = moneyDebounce.amount + delta
            if moneyDebounce.timer then
                task.cancel(moneyDebounce.timer)
            end
            moneyDebounce.timer = task.delay(5, FlushMoney)
        end)
    end
end)

local EspFruitsSection = EspTab:Section({Name = "Fruits", Side = 1, Icon = "cherry"})

-- ─────────────────────────────────────────────────────────
-- DROPPED FRUITS ESP SYSTEM
-- Target: Workspace direct children
-- Filter: Name contains "Dropped" + has Handle.ProximityPrompt
-- Color: Dynamic from Handle.BrickColor
-- ─────────────────────────────────────────────────────────
do
    local FruitEspConfig = {
        BoxThickness  = 1.5,
        TextSize      = 13,
        Font          = Drawing.Fonts.Plex,
        MaxDistance   = 10000,
        DefaultColor  = Color3.fromRGB(255, 255, 255),
    }

    local FruitEspCache       = {}  -- [Fruit] = { Objects = {}, CleanName = "" }
    local FruitRenderConn     = nil
    local FruitChildAddedConn = nil
    local FruitChildRemovedConn = nil

    -- Clean fruit name ONCE at creation time, never during RenderStepped
    local function CleanFruitName(raw)
        local name = raw
        name = string.gsub(name, "_Dropped", "")
        name = string.gsub(name, "Dropped ", "")
        name = string.gsub(name, "Dropped",  "")
        name = string.gsub(name, "_+$",      "")
        name = string.gsub(name, "^%s+",     "")
        name = string.gsub(name, "%s+$",     "")
        return string.upper(name)
    end

    local function IsFruit(Object)
        if not (Object and Object.Name and string.find(Object.Name, "Dropped")) then return false end
        local Handle = Object:FindFirstChild("Handle")
        if not Handle then return false end
        return Handle:FindFirstChildWhichIsA("ProximityPrompt") ~= nil
    end

    local function CreateFruitESP(Fruit)
        if FruitEspCache[Fruit] then return end

        local Objects = {
            Box          = Drawing.new("Square"),
            BoxOutline   = Drawing.new("Square"),
            Tracer       = Drawing.new("Line"),
            NameText     = Drawing.new("Text"),
            DistText     = Drawing.new("Text"),
        }

        Objects.Box.Thickness        = FruitEspConfig.BoxThickness
        Objects.Box.Filled           = false
        Objects.Box.ZIndex           = 3
        Objects.Box.Visible          = false

        Objects.BoxOutline.Color     = Color3.new(0, 0, 0)
        Objects.BoxOutline.Thickness = FruitEspConfig.BoxThickness + 1.5
        Objects.BoxOutline.Filled    = false
        Objects.BoxOutline.ZIndex    = 2
        Objects.BoxOutline.Visible   = false

        Objects.Tracer.Thickness     = 1
        Objects.Tracer.Transparency  = 0.6
        Objects.Tracer.Visible       = false

        Objects.NameText.Font        = FruitEspConfig.Font
        Objects.NameText.Size        = FruitEspConfig.TextSize
        Objects.NameText.Center      = true
        Objects.NameText.Outline     = true
        Objects.NameText.Visible     = false

        Objects.DistText.Font        = FruitEspConfig.Font
        Objects.DistText.Size        = FruitEspConfig.TextSize - 1
        Objects.DistText.Color       = Color3.fromRGB(255, 255, 255)
        Objects.DistText.Center      = true
        Objects.DistText.Outline     = true
        Objects.DistText.Visible     = false

        FruitEspCache[Fruit] = {
            Objects   = Objects,
            CleanName = CleanFruitName(Fruit.Name),
        }
    end

    local function RemoveFruitESP(Fruit)
        local entry = FruitEspCache[Fruit]
        if entry then
            for _, DrawingObj in pairs(entry.Objects) do
                DrawingObj:Remove()
            end
            FruitEspCache[Fruit] = nil
        end
    end

    local function ClearAllFruitESP()
        for Fruit, _ in pairs(FruitEspCache) do
            RemoveFruitESP(Fruit)
        end
        FruitEspCache = {}
    end

    local function FruitUpdate()
        for Fruit, entry in pairs(FruitEspCache) do
            local Objects = entry.Objects

            if Fruit and Fruit.Parent == Workspace and Fruit:FindFirstChild("Handle") then
                local Handle     = Fruit.Handle
                local Position   = Handle.Position
                local ScreenPos, OnScreen = Camera:WorldToViewportPoint(Position)
                local Distance   = (Camera.CFrame.Position - Position).Magnitude

                if OnScreen and Distance <= FruitEspConfig.MaxDistance then
                    local FruitColor = Handle.BrickColor.Color or FruitEspConfig.DefaultColor

                    local Extents = Handle.Size
                    local Scale   = Camera.ViewportSize.Y / (Distance * 2 * math.tan(math.rad(Camera.FieldOfView / 2)))
                    local SizeX   = math.max(Extents.X * Scale * 1.5, 12)
                    local SizeY   = math.max(Extents.Y * Scale * 1.5, 12)
                    local BoxPos  = Vector2.new(ScreenPos.X - SizeX / 2, ScreenPos.Y - SizeY / 2)

                    Objects.Box.Size       = Vector2.new(SizeX, SizeY)
                    Objects.Box.Position   = BoxPos
                    Objects.Box.Color      = FruitColor
                    Objects.Box.Visible    = true

                    Objects.BoxOutline.Size     = Vector2.new(SizeX, SizeY)
                    Objects.BoxOutline.Position = BoxPos
                    Objects.BoxOutline.Visible  = true

                    Objects.Tracer.From    = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                    Objects.Tracer.To      = Vector2.new(ScreenPos.X, BoxPos.Y + SizeY)
                    Objects.Tracer.Color   = FruitColor
                    Objects.Tracer.Visible = true

                    Objects.NameText.Text     = entry.CleanName
                    Objects.NameText.Position = Vector2.new(ScreenPos.X, BoxPos.Y - 16)
                    Objects.NameText.Color    = FruitColor
                    Objects.NameText.Visible  = true

                    Objects.DistText.Text     = string.format("[%d m]", math.floor(Distance))
                    Objects.DistText.Position = Vector2.new(ScreenPos.X, BoxPos.Y + SizeY + 4)
                    Objects.DistText.Visible  = true
                else
                    for _, Obj in pairs(Objects) do Obj.Visible = false end
                end
            else
                RemoveFruitESP(Fruit)
            end
        end
    end

    local function FruitScan()
        for _, Child in ipairs(Workspace:GetChildren()) do
            if IsFruit(Child) then
                CreateFruitESP(Child)
            end
        end
    end

    function FruitEsp_Enable()
        FruitScan()

        FruitChildAddedConn = Workspace.ChildAdded:Connect(function(Child)
            task.wait(0.2)
            if Settings.FruitEspEnabled and IsFruit(Child) then
                CreateFruitESP(Child)
            end
        end)

        FruitChildRemovedConn = Workspace.ChildRemoved:Connect(function(Child)
            RemoveFruitESP(Child)
        end)

        FruitRenderConn = RunService.RenderStepped:Connect(FruitUpdate)
    end

    function FruitEsp_Disable()
        if FruitRenderConn      then FruitRenderConn:Disconnect();      FruitRenderConn      = nil end
        if FruitChildAddedConn  then FruitChildAddedConn:Disconnect();  FruitChildAddedConn  = nil end
        if FruitChildRemovedConn then FruitChildRemovedConn:Disconnect(); FruitChildRemovedConn = nil end
        ClearAllFruitESP()
    end
end

    local function InteractWithMerchant(method)
        local npcsFolder = Workspace:FindFirstChild("ServiceNPCs")
        local merchant   = npcsFolder and npcsFolder:FindFirstChild("MerchantNPC")
        if not merchant then return false end

        -- Teleport to merchant
        local ok, cf = pcall(function() return merchant:GetModelCFrame() end)
        if ok and cf then
            TeleportTo(cf)
            task.wait(0.5)
        end

        -- Find prompt
        local hrp    = merchant:FindFirstChild("HumanoidRootPart")
        local prompt = hrp and hrp:FindFirstChild("MerchantPrompt")
        if not prompt then return false end

        if method == "Executor" then
            pcall(function() fireproximityprompt(prompt) end)
        else
            -- Luau safe interaction
            pcall(function()
                local prev = prompt.HoldDuration
                local prevLOS = prompt.RequiresLineOfSight
                prompt.HoldDuration = 0
                prompt.RequiresLineOfSight = false
                prompt:InputHoldBegin()
                task.wait(0.1)
                prompt:InputHoldEnd()
                prompt.HoldDuration = prev
                prompt.RequiresLineOfSight = prevLOS
            end)

        task.wait(0.8) -- wait for UI to open
        return true

        local pg     = LocalPlayer:FindFirstChild("PlayerGui")
        local holder = pg
            and pg:FindFirstChild("MerchantUI")
            and pg.MerchantUI:FindFirstChild("MainFrame")
            and pg.MerchantUI.MainFrame:FindFirstChild("Frame")
            and pg.MerchantUI.MainFrame.Frame:FindFirstChild("Content")
            and pg.MerchantUI.MainFrame.Frame.Content:FindFirstChild("Holder")

                    seen[c] = true

                for code in block:gmatch("<strong>(.-)</strong>") do addCode(code) end
            end
            for code in body:gmatch("<strong>(.-)</strong>") do addCode(code) end
            for code in body:gmatch("<code>(.-)</code>") do addCode(code) end
            for code in body:gmatch('data%-code="([^"]+)"') do addCode(code) end
            for code in body:gmatch("<li>%s*([%w%!%-%_]+)%s*[%-–—]") do addCode(code) end

            local total = #codes
            if total == 0 then
                RedeemStatusParagraph:SetText(RedeemStatusError("No codes found on page"))
                Library:Notification({ Title = "Code Redeemer", Description = "Couldn't parse any codes from the page.", Duration = 5 })

                    return CodeRedeem:InvokeServer(code)

            })

                local name = obj.Name
                name = name:gsub("_Dropped", ""):gsub("Dropped ", ""):gsub("Dropped", "")
                name = name:gsub("_+$", ""):gsub("^%s+", ""):gsub("%s+$", "")
                if name == "" then name = obj.Name end
                if not counts[name] then
                    counts[name] = 0
                    table.insert(order, name)
                end
                counts[name] = counts[name] + 1
            end
        end
    end

    if #order == 0 then
        ActiveFruitsParagraph:SetText("Status: No fruits spawned in this server.")
        return
    end

    table.sort(order)

    local lines = {"Fruits in Server:"}
    for _, name in ipairs(order) do
        local count = counts[name]
        if count > 1 then
            table.insert(lines, "• " .. count .. "x " .. name)
        else
            table.insert(lines, "• " .. name)
        end
    end

    ActiveFruitsParagraph:SetText(table.concat(lines, "\n"))
end

        task.wait(4)
    end
end)

task.spawn(function()
    -- Multiple Farm At Time: cache and timing state
    local grindMobList = {}
    local grindMobIndex = 1
    local grindLastRefresh = 0
    local grindRefreshInterval = 2
    -- Cancel Quest On Switch debounce
    local _lastAbandonTick = 0

    -- ── Death-recovery state ──────────────────────────────
    -- lastFarmCFrame: updated every tick while actively farming a mob.
    --   Stores the mob's world position so we can return there after death.
    -- _diedRecently: flipped true the moment the player is detected dead
    --   inside any farm-loop repeat.  Cleared once mobs are found again
    --   OR once the recovery teleport fires.
    local lastFarmCFrame = nil
    local _diedRecently  = false

    -- Helper: call at every "mob not found" path.
    -- If the player just died and we have a stored position, teleport there
    -- so streaming re-loads the mob cluster, then let the loop resume normally.
    local function TryDeathRecovery(mobName)
        if not (_diedRecently and lastFarmCFrame) then return end
        _diedRecently = false
        local pChar0 = LocalPlayer.Character
        local pHrp0  = pChar0 and pChar0:FindFirstChild("HumanoidRootPart")
        if not pHrp0 then return end
        FarmStatus:SetText(BuildFarmStatus("↩ Returning to last position", mobName, nil, nil, nil))
        TeleportTo(lastFarmCFrame)
    end

    while task.wait() do
        if Settings.AutoLevel and ActiveAutomation == "farm" then
            local pChar = LocalPlayer.Character
            local pHum = pChar and pChar:FindFirstChild("Humanoid")
            if pChar and pHum and pHum.Health > 0 then
                HandleAutoEquip()
            end

            local bestID, bestData = GetBestQuest()

            if bestData then
                local mobName = "Unknown"
                local reqLevel = 0

                if bestData.requirements and bestData.requirements[1] then
                    mobName = bestData.requirements[1].npcType or mobName
                end

Level:</b></font> <font color='#DDDDDD'>"..tostring(reqLevel).."</font>")

                        --   2. Fire TeleportToPortal remote ONCE for that island
                        --   3. Wait 0.5s for portal to land
                        --   4. Snap to mob farming position
                        --   5. Kill loop — ONLY mob TP inside, NO portal re-fire
                        --   6. Mob dead → outer loop acquires next mob → repeat from 2

                        -- Firing the portal remote every tick was the bug causing
                        -- the infinite portal loop.  It must fire exactly once when
                        -- switching to a new mob / island.
                        -- ─────────────────────────────────────────────────────────
                        local targetMob = GetAliveMob(mobName)

                        if targetMob then
                            Settings.CurrentTarget = targetMob
                            local humanoid = targetMob:FindFirstChild("Humanoid")
                            local hrp      = targetMob:FindFirstChild("HumanoidRootPart")

                            if humanoid and hrp then

                                local islandName = GetClosestPortalIsland(hrp.Position)

                                    end

                                    pcall(function()

                                        TeleportTo(GetTeleportCFrame(hrp))

                                    task.wait()
                                until humanoid.Health <= 0
                                    or not targetMob.Parent
                                    or not hasQuestActive
                                    or not Settings.AutoLevel
                                    or ActiveAutomation ~= "farm"
                            end
                            StopMovement()
                            Settings.CurrentTarget = nil
                        else
                            local progressText = "Unknown"
                            pcall(function()
                                progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                            end)

                            TryDeathRecovery(mobName)
                            FarmStatus:SetText(BuildFarmStatus("Searching", mobName, nil, nil, progressText))

                    -- DEFAULT STYLE (original untouched logic)

                    elseif Settings.AutoFarmStyle == "Default" then

                            local progressText = "Unknown"
                            pcall(function()
                                progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text

                            Settings.CurrentTarget = nil
                            TryDeathRecovery(mobName)

                    -- GRIND ALL MOBS STYLE
                    -- ─────────────────────────────────────────
                    elseif Settings.AutoFarmStyle == "Multiple Farm At Time" then
                        local now = tick()

                        -- Rebuild mob list only on interval or when empty
                        if now - grindLastRefresh >= grindRefreshInterval or #grindMobList == 0 then

                            local npcsFolder = Workspace:FindFirstChild("NPCs")
                            grindMobList = {}
                            grindMobIndex = 1
                            if npcsFolder then
                                local targetMatch = CleanString(mobName)

                                            end
                                        end
                                    end
                                end
                            end
                        end

                        -- Fast purge: backwards sweep, remove dead/invalid entries

                        for i = count, 1, -1 do

                            local e = grindMobList[1]
                            local hum, hrp = e.hum, e.hrp
                            if hum.Health > 0 and e.model.Parent then
                                Settings.CurrentTarget = e.model

                                    progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                                end)
                                FarmStatus:SetText(BuildFarmStatus("Finishing", mobName, hum.Health, hum.MaxHealth, progressText))

                                grindMobIndex = 1

                            if grindMobIndex > count then grindMobIndex = 1 end
                            local e = grindMobList[grindMobIndex]
                            local hum, hrp = e.hum, e.hrp

                            if not e.model.Parent or hum.Health <= 0 then
                                table.remove(grindMobList, grindMobIndex)

                                end

                                task.wait()

                                -- Instant switch check: if hp dropped, move on immediately
                                -- If hp unchanged and mob still alive, mark as immune and remove
                                if hum.Parent and hum.Health > 0 and hum.Health >= prevHp then
                                    table.remove(grindMobList, grindMobIndex)
                                else
                                    grindMobIndex = grindMobIndex + 1

                    -- alive mobs with this name.  Falls back to
                    -- the Default single-target logic when only
                    -- one mob is present.
                    -- ─────────────────────────────────────────
                    elseif Settings.AutoFarmStyle == "Mid Point" then
                        local midCF = GetMidPointCFrame(mobName)

                        if midCF then
                            -- Multiple mobs found → teleport to their collective centre
                            -- Store mid position as last farm location
                            lastFarmCFrame = midCF
                            local progressText = "Unknown"
                            pcall(function()
                                progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                            end)
                            FarmStatus:SetText(BuildFarmStatus("Mid-Point Farm", mobName, nil, nil, progressText))
                            TeleportTo(midCF)
                            hasQuestActive = questUI and questUI:FindFirstChild("Quest") and questUI.Quest.Visible
                            task.wait()

                        else
                            -- Single mob or none → behave identically to Default style
                            local targetMob = GetAliveMob(mobName)
                            if targetMob then
                                Settings.CurrentTarget = targetMob
                                local humanoid = targetMob:FindFirstChild("Humanoid")
                                local hrp = targetMob:FindFirstChild("HumanoidRootPart")
                                if humanoid and hrp then
                                    repeat
                                        if not Settings.AutoLevel then break end
                                        pChar = LocalPlayer.Character
                                        pHum  = pChar and pChar:FindFirstChild("Humanoid")
                                        if not pChar or not pHum or pHum.Health <= 0 then
                                            _diedRecently = true
                                            StopMovement()
                                            Settings.CurrentTarget = nil
                                            local newChar = LocalPlayer.CharacterAdded:Wait()
                                            newChar:WaitForChild("Humanoid", 10)
                                            newChar:WaitForChild("HumanoidRootPart", 10)
                                            task.wait(1.5)
                                            HandleAutoEquip()
                                            break
                                        end
                                        HandleAutoEquip()
                                        if hrp and hrp.Parent then
                                            lastFarmCFrame = CFrame.new(hrp.Position)
                                        end
                                        local progressText = "Unknown"
                                        pcall(function()
                                            progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                                        end)
                                        FarmStatus:SetText(BuildFarmStatus("Killing", mobName, humanoid.Health, humanoid.MaxHealth, progressText))
                                        TeleportTo(GetTeleportCFrame(hrp))
                                        hasQuestActive = questUI and questUI:FindFirstChild("Quest") and questUI.Quest.Visible
                                        task.wait()
                                    until humanoid.Health <= 0
                                        or not targetMob.Parent
                                        or not hasQuestActive
                                        or not Settings.AutoLevel
                                        or ActiveAutomation ~= "farm"
                                end
                                StopMovement()
                                Settings.CurrentTarget = nil
                            else
                                local progressText = "Unknown"
                                pcall(function()
                                    progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                                end)
                                Settings.CurrentTarget = nil
                                TryDeathRecovery(mobName)
                                FarmStatus:SetText(BuildFarmStatus("Searching", mobName, nil, nil, progressText))
                            end
                        end

                    -- ─────────────────────────────────────────
                    -- ZIG ZAG STYLE
                    -- Performs smart randomised offset teleports
                    -- around the target mob each tick, cycling
                    -- through 12 directional slots and varying
                    -- the distance slightly to look organic.
                    -- ─────────────────────────────────────────
                    elseif Settings.AutoFarmStyle == "Zig Zag" then
                        local targetMob = GetAliveMob(mobName)

                        if targetMob then
                            Settings.CurrentTarget = targetMob
                            local humanoid = targetMob:FindFirstChild("Humanoid")
                            local hrp      = targetMob:FindFirstChild("HumanoidRootPart")

                            if humanoid and hrp then
                                -- zigState: persists for the lifetime of this mob engagement.
                                -- slot=0 is the uninitialised sentinel; GetZigZagCFrame
                                -- bootstraps all fields on the very first call.
                                local zigState = {
                                    slot        = 0,
                                    prevSlot    = 0,
                                    lockedUntil = 0,
                                    jitter      = 0,
                                }

                                repeat
                                    if not Settings.AutoLevel then break end

                                    pHum  = pChar and pChar:FindFirstChild("Humanoid")
                                    if not pChar or not pHum or pHum.Health <= 0 then
                                        _diedRecently = true
                                        StopMovement()
                                        Settings.CurrentTarget = nil
                                        local newChar = LocalPlayer.CharacterAdded:Wait()
                                        newChar:WaitForChild("Humanoid", 10)
                                        newChar:WaitForChild("HumanoidRootPart", 10)
                                        task.wait(1.5)
                                        HandleAutoEquip()
                                        break
                                    end

                                    HandleAutoEquip()

                                    -- Update death-recovery anchor every tick
                                    if hrp and hrp.Parent then
                                        lastFarmCFrame = CFrame.new(hrp.Position)
                                    end

                                    local progressText = "Unknown"
                                    pcall(function()
                                        progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                                    end)

                                    FarmStatus:SetText(BuildFarmStatus("Zig-Zag Farm", mobName, humanoid.Health, humanoid.MaxHealth, progressText))

                                    -- Recalculate offset relative to mob's CURRENT position
                                    -- so the pattern follows a moving mob naturally
                                    TeleportTo(GetZigZagCFrame(hrp, zigState))

                                    hasQuestActive = questUI and questUI:FindFirstChild("Quest") and questUI.Quest.Visible
                                    task.wait()
                                until humanoid.Health <= 0
                                    or not targetMob.Parent
                                    or not hasQuestActive
                                    or not Settings.AutoLevel
                                    or ActiveAutomation ~= "farm"
                            end
                            StopMovement()
                            Settings.CurrentTarget = nil
                        else
                            local progressText = "Unknown"
                            pcall(function()
                                progressText = questUI.Quest.Quest.Holder.Content.QuestInfo.QuestRequirement.Text
                            end)
                            Settings.CurrentTarget = nil
                            TryDeathRecovery(mobName)
                            FarmStatus:SetText(BuildFarmStatus("Searching", mobName, nil, nil, progressText))
                        end
                    end  -- end of AutoFarmStyle branch

                else
                    -- Quest not active — claim it
                    grindMobList = {}
                    Settings.CurrentTarget = nil
                    FarmStatus:SetText(BuildFarmStatus("Claiming Quest", mobName, nil, nil, nil))

                    local questNpc = Workspace:FindFirstChild(bestID, true)
                    if questNpc then
                        TeleportTo(questNpc:GetPivot())
                        task.wait(0.5)
                        local args = {tostring(bestID)}
                        QuestAcceptRemote:FireServer(unpack(args))
                        task.wait(0.5)
                    end
                end
            end
        else
            -- Disabled or yielding to higher priority
            grindMobList = {}
            grindMobIndex = 1
            if Settings.CurrentTarget then
                Settings.CurrentTarget = nil
            end
        end
    end
end)

task.spawn(function()
    -- Cached reference to the ability remote (resolved once)
    local RequestAbilityRemote = ReplicatedStorage
        :WaitForChild("AbilitySystem")
        :WaitForChild("Remotes")
        :WaitForChild("RequestAbility")

    local function FireSkillSet(flagKey)
        local selectedSkills = Library.Flags[flagKey]
        if not selectedSkills then return end
        local char = LocalPlayer.Character
        local hum  = char and char:FindFirstChild("Humanoid")
        if not char or not hum or hum.Health <= 0 then return end
        for k, v in pairs(selectedSkills) do
            local skillNum
            if type(k) == "string" and v == true then
                skillNum = tonumber(k)
            elseif type(k) == "number" and type(v) == "string" then
                skillNum = tonumber(v)

-- ─────────────────────────────────────────────────────────
-- SPECIFIC FARM / CLOSEST FARM LOOP (with optional quest integration)
-- ─────────────────────────────────────────────────────────
task.spawn(function()
    -- ── Helpers ──────────────────────────────────────────

    -- Reads multi-select dropdown flag safely (handles both
    -- {["Name"] = true} and {[1] = "Name"} library formats)
    local function GetSelectedMobNames()
        local raw = Library.Flags.SpecificFarmMobsDropdown
        if not raw then return {} end
        local names = {}
        for k, v in pairs(raw) do
            local mobName
            if type(k) == "string" and v == true and k ~= "No mobs found" then
                mobName = k
            elseif type(k) == "number" and type(v) == "string" and v ~= "No mobs found" then
                mobName = v
            end
            if mobName then
                table.insert(names, CleanString(mobName))
            end
        end
        return names
    end

    local function FindClosestMob(nameFilter)
        local npcsFolder = Workspace:FindFirstChild("NPCs")
        if not npcsFolder then return nil end
        local pChar = LocalPlayer.Character
        local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")
        if not pHrp then return nil end
        local best     = nil
        local bestDist = math.huge
        for _, mob in ipairs(npcsFolder:GetChildren()) do
            if mob:IsA("Model") then
                if mob.Name == "TrainingDummy" then continue end

                local hum = mob:FindFirstChild("Humanoid")
                if not hum or hum.Health <= 0 then continue end

                local pass = true
                if nameFilter and #nameFilter > 0 then
                    local baseName = CleanString(string.gsub(mob.Name, "%d+$", ""))
                    local dispName = CleanString(string.gsub(hum.DisplayName or "", "%s*%[.-%]$", ""))
                    pass = false
                    for _, n in ipairs(nameFilter) do
                        if baseName == n or dispName == n then
                            pass = true
                            break
                        end
                    end
                end
                if not pass then continue end

                local hrp = mob:FindFirstChild("HumanoidRootPart")
                local dist
                if hrp then
                    dist = (pHrp.Position - hrp.Position).Magnitude
                else
                    local ok, cf = pcall(function() return mob:GetModelCFrame() end)
                    dist = ok and cf and (pHrp.Position - cf.Position).Magnitude or math.huge
                end

                if dist < bestDist then
                    bestDist = dist
                    best     = mob
                end
            end
        end
        return best
    end

    -- ── Quest cache: cleanedMobName -> questID (built once) ──
    local mobQuestCache = {}
    do
        local searchAreas = {QuestModule}
        if QuestModule.RepeatableQuests then
            table.insert(searchAreas, QuestModule.RepeatableQuests)
        end
        for _, area in ipairs(searchAreas) do
            for id, data in pairs(area) do
                if type(data) == "table" and data.requirements then
                    for _, req in ipairs(data.requirements) do
                        local npc = req.npcType
                        if npc then
                            mobQuestCache[CleanString(npc)] = tostring(id)
                        end
                    end
                end
            end
        end
    end

    -- Returns questID string for a mob model, or nil if none found
    local function GetQuestForMob(mobModel)
        if not mobModel then return nil end
        local hum      = mobModel:FindFirstChild("Humanoid")
        local baseName = CleanString(string.gsub(mobModel.Name, "%d+$", ""))
        local dispName = hum and CleanString(string.gsub(hum.DisplayName or "", "%s*%[.-%]$", "")) or ""
        return mobQuestCache[baseName] or mobQuestCache[dispName]
    end

    -- Lightweight quest-active check via QuestUI visibility
    local function IsQuestActive()
        local gui = LocalPlayer:FindFirstChild("PlayerGui")
        local qUI = gui and gui:FindFirstChild("QuestUI")
        return qUI and qUI:FindFirstChild("Quest") and qUI.Quest.Visible
    end

    -- ── Quest state (per-session, not per-frame) ──────────
    local lastClaimedQuestID = nil   -- ID currently claimed / being farmed
    local claimCooldown      = 0     -- tick() timestamp before which we won't re-claim
    local CLAIM_COOLDOWN_SEC = 4     -- seconds between claim attempts

    -- ── Main loop ─────────────────────────────────────────
    while true do
        local specificOn = Settings.SpecificFarmEnabled
        local closestOn  = Settings.ClosestFarmEnabled

        if not specificOn and not closestOn then
            Settings.SpecificFarmCurrentTarget = nil
            lastClaimedQuestID = nil
            task.wait(0.5)
            continue
        end

        if specificOn and ActiveAutomation ~= "specific" then
            Settings.SpecificFarmCurrentTarget = nil
            task.wait(0.3)
            continue
        end

        if closestOn and not specificOn and ActiveAutomation ~= "closest" then
            Settings.SpecificFarmCurrentTarget = nil
            task.wait(0.3)
            continue
        end

        -- Validate player alive
        local pChar = LocalPlayer.Character
        local pHum  = pChar and pChar:FindFirstChild("Humanoid")
        if not pChar or not pHum or pHum.Health <= 0 then
            Settings.SpecificFarmCurrentTarget = nil
            task.wait(1)
            continue
        end

        -- Find target
        local current = nil
        if specificOn then
            if Settings.AnyMobEnabled then
                -- Any Mob: skip name filter entirely, find closest valid mob
                current = FindClosestMob(nil)
            else
                local selectedNames = GetSelectedMobNames()
                if #selectedNames > 0 then
                    current = FindClosestMob(selectedNames)
                end
            end
        end
        if not current and closestOn then
            current = FindClosestMob(nil)
        end

        if not current then
            Settings.SpecificFarmCurrentTarget = nil
            task.wait(0.3)
            continue
        end

        -- ── Quest logic: claim before engaging ──────────────
        if Settings.SpecificFarmQuestEnabled then
            local questID = GetQuestForMob(current)
            if questID and not IsQuestActive() then
                if tick() >= claimCooldown then
                    claimCooldown = tick() + CLAIM_COOLDOWN_SEC
                    local questNpc = Workspace:FindFirstChild(questID, true)
                    if questNpc then
                        local ok, pivot = pcall(function() return questNpc:GetPivot() end)
                        if ok and pivot then
                            TeleportTo(pivot)
                            task.wait(0.4)
                        end
                    end
                    pcall(function()
                        QuestAcceptRemote:FireServer(questID)
                    end)
                    lastClaimedQuestID = questID
                    task.wait(0.4)
                end
                task.wait(0.1)
                continue
            end
        end

        -- ── Engage ──
        Settings.SpecificFarmCurrentTarget = current
        HandleAutoEquip()

        local humanoid = current:FindFirstChild("Humanoid")
        if not humanoid or humanoid.Health <= 0 then
            task.wait(0.1)
            continue
        end

        local hrp = current:FindFirstChild("HumanoidRootPart")
        if not hrp then
            local ok, cf = pcall(function() return current:GetModelCFrame() end)
            if ok and cf then
                local pChar0 = LocalPlayer.Character
                local pHrp0  = pChar0 and pChar0:FindFirstChild("HumanoidRootPart")
                if pHrp0 then pHrp0.CFrame = cf end
            end
            local waited = 0
            repeat
                task.wait(0.1)
                waited = waited + 0.1
                hrp = current:FindFirstChild("HumanoidRootPart")
            until hrp or waited >= 3 or not current.Parent
        end

        if not hrp or not current.Parent or humanoid.Health <= 0 then
            task.wait(0.1)
            continue
        end

        if Settings.AutoFarmStyle == "Gate Farming" then
            local islandName = GetClosestPortalIsland(hrp.Position)
            if islandName then
                pcall(function()
                    TeleportToPortalRemote:FireServer(islandName)
                end)
                task.wait(0.5)
            end
            if hrp and hrp.Parent then
                TeleportTo(GetTeleportCFrame(hrp))
            end
        end

        repeat
            if not Settings.SpecificFarmEnabled and not Settings.ClosestFarmEnabled then break end

            pChar = LocalPlayer.Character
            pHum  = pChar and pChar:FindFirstChild("Humanoid")
            if not pChar or not pHum or pHum.Health <= 0 then
                StopMovement()
                Settings.SpecificFarmCurrentTarget = nil
                local newChar = LocalPlayer.CharacterAdded:Wait()
                newChar:WaitForChild("Humanoid", 10)
                newChar:WaitForChild("HumanoidRootPart", 10)
                task.wait(1.5)
                HandleAutoEquip()
                break
            end

            HandleAutoEquip()

            if specificOn and Settings.AutoFarmStyle == "Mid Point" then
                local baseName = string.gsub(current.Name, "%d+$", "")
                local midCF    = GetMidPointCFrame(baseName)
                if midCF then
                    TeleportTo(midCF)
                else
                    TeleportTo(GetTeleportCFrame(hrp))
                end
            else
                TeleportTo(GetTeleportCFrame(hrp))
            end

            task.wait()
        until humanoid.Health <= 0
            or not current.Parent
            or (not Settings.SpecificFarmEnabled and not Settings.ClosestFarmEnabled)

        StopMovement()
        Settings.SpecificFarmCurrentTarget = nil
    end
end)

do
    local GuiService        = game:GetService("GuiService")

    end

    local function IsDungeonActive()
        local dUI = GetDungeonUI()
        if not dUI then return false end
        if dUI.Enabled then return true end
        local cf = dUI:FindFirstChild("ContentFrame")
        if cf and cf.Visible then
            local df = cf:FindFirstChild("DungeonFrame")

        end

    local function ClickButton(button)
        if not button then return end
        GuiService.SelectedObject = button
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(true,  13, false, game)
        task.wait(0.1)
        VirtualInputManager:SendKeyEvent(false, 13, false, game)
        task.wait(0.2)
        GuiService.SelectedObject = nil
    end

    local function FindDungeonMob()
        local npcsFolder = Workspace:FindFirstChild("NPCs")
        if not npcsFolder then return nil end
        local pChar = LocalPlayer.Character
        local pHrp  = pChar and pChar:FindFirstChild("HumanoidRootPart")
        if not pHrp then return nil end

        local style = Settings.DungeonTargetStyle
        local best      = nil
        local bestScore = nil  -- meaning depends on style

        for _, mob in ipairs(npcsFolder:GetChildren()) do
            if mob:IsA("Model") then
                local isDungeon = mob:GetAttribute("DungeonNPC") or mob:GetAttribute("DungeonBoss")
                if isDungeon then
                    local hum = mob:FindFirstChild("Humanoid")
                    local hrp = mob:FindFirstChild("HumanoidRootPart")
                    if hum and hrp and hum.Health > 0 then
                        local score

                        if style == "Closest" then
                            score = (pHrp.Position - hrp.Position).Magnitude
                            if bestScore == nil or score < bestScore then
                                bestScore = score
                                best = mob
                            end

                        elseif style == "Strongest" then
                            local dmg = mob:GetAttribute("BaseDamage")
                            if dmg then
                                if bestScore == nil or dmg > bestScore then
                                    bestScore = dmg
                                    best = mob
                                end
                            end

                        elseif style == "Weakest" then
                            local dmg = mob:GetAttribute("BaseDamage")
                            if dmg then
                                if bestScore == nil or dmg < bestScore then
                                    bestScore = dmg
                                    best = mob
                                end
                            end

                        elseif style == "Lowest HP" then
                            score = hum.Health
                            if bestScore == nil or score < bestScore then
                                bestScore = score
                                best = mob
                            end

                        elseif style == "Boss Focus" then
                            if mob:GetAttribute("DungeonBoss") == true then
                                score = (pHrp.Position - hrp.Position).Magnitude
                                if bestScore == nil or score < bestScore then
                                    bestScore = score
                                    best = mob
                                end
                            end
                        end
                    end
                end
            end
        end

        -- Fallback to Closest if the attribute-based styles found nothing
        if not best and (style == "Strongest" or style == "Weakest" or style == "Boss Focus") then
            local bestDist = math.huge
            for _, mob in ipairs(npcsFolder:GetChildren()) do
                if mob:IsA("Model") then
                    local isDungeon = mob:GetAttribute("DungeonNPC") or mob:GetAttribute("DungeonBoss")
                    if isDungeon then
                        local hum = mob:FindFirstChild("Humanoid")
                        local hrp = mob:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 then
                            local dist = (pHrp.Position - hrp.Position).Magnitude
                            if dist < bestDist then
                                bestDist = dist
                                best = mob
                            end
                        end
                    end
                end
            end
        end

        return best
    end

                    and replayFrame.Holder:FindFirstChild("Button")
                    and replayFrame.Holder.Button:FindFirstChild("ReplayDungeonButton")

        end
    end)

    -- ── Auto Dungeon farm loop ────────────────────────────
    task.spawn(function()
        while true do
            if not Settings.AutoDungeon then
                Settings.DungeonCurrentTarget = nil
                Settings.DungeonDistanceOverride = nil
                task.wait(0.5)
                continue
            end

            if ActiveAutomation ~= "dungeon" then
                Settings.DungeonCurrentTarget = nil
                task.wait(0.3)
                continue
            end

            -- Validate player alive
            local pChar = LocalPlayer.Character
            local pHum  = pChar and pChar:FindFirstChild("Humanoid")
            if not pChar or not pHum or pHum.Health <= 0 then
                Settings.DungeonCurrentTarget = nil
                task.wait(1)
                continue
            end

            local current = FindDungeonMob()

            if not current then
                Settings.DungeonCurrentTarget = nil
                task.wait(0.3)
                continue
            end

            Settings.DungeonCurrentTarget = current
            HandleAutoEquip()

            -- Auto Position: use 10-stud distance for boss mobs, restore for normal mobs
            if Settings.DungeonAutoPosition then
                local bossObj = current:FindFirstChild("Boss")
                local isBoss  = bossObj and bossObj.Value == true
                if isBoss and Settings.TpDistance < 10 then
                    Settings.DungeonDistanceOverride = 10
                else
                    Settings.DungeonDistanceOverride = nil
                end
            else
                Settings.DungeonDistanceOverride = nil
            end

            local humanoid = current:FindFirstChild("Humanoid")
            local hrp      = current:FindFirstChild("HumanoidRootPart")

            if humanoid and hrp then
                repeat
                    if not Settings.AutoDungeon then break end

                    pChar = LocalPlayer.Character
                    pHum  = pChar and pChar:FindFirstChild("Humanoid")
                    if not pChar or not pHum or pHum.Health <= 0 then
                        StopMovement()
                        Settings.DungeonCurrentTarget = nil
                        local newChar = LocalPlayer.CharacterAdded:Wait()
                        newChar:WaitForChild("Humanoid", 10)
                        newChar:WaitForChild("HumanoidRootPart", 10)
                        task.wait(1.5)
                        HandleAutoEquip()
                        break
                    end

                    HandleAutoEquip()

                    -- Mid Point style: teleport to the centre of all matching dungeon mobs
                    if Settings.AutoFarmStyle == "Mid Point" then
                        local baseName = string.gsub(current.Name, "%d+$", "")
                        local midCF    = GetMidPointCFrame(baseName)
                        if midCF then
                            TeleportTo(midCF)
                        else
                            TeleportTo(GetTeleportCFrame(hrp))
                        end
                    else
                        TeleportTo(GetTeleportCFrame(hrp))
                    end

                    task.wait()
                until humanoid.Health <= 0
                    or not current.Parent
                    or not Settings.AutoDungeon
            end

            StopMovement()
            Settings.DungeonCurrentTarget = nil
            Settings.DungeonDistanceOverride = nil
        end
    end)
end

local function GetLowestLevelMob()
    local npcsFolder = Workspace:FindFirstChild("NPCs")
    if not npcsFolder then return nil end
    local lowestMob = nil
    local lowestLevel = math.huge
    for _, mob in ipairs(npcsFolder:GetChildren()) do
        if mob:IsA("Model") then
            local humanoid = mob:FindFirstChild("Humanoid")
            local hrp = mob:FindFirstChild("HumanoidRootPart")
            if humanoid and hrp and humanoid.Health > 0 then
                local displayName = humanoid.DisplayName or ""
                local level = tonumber(string.match(displayName, "%[Lv%.(%d+)%]"))
                if level and level < lowestLevel then
                    lowestLevel = level
                    lowestMob = mob
                end
            end
        end
    end
    return lowestMob
end

-- PRIORITY SCHEDULER
-- ─────────────────────────────────────────────────────────
task.spawn(function()
    -- Pre-build priority sort order (static unless user changes it)
    local function BuildSortedPriority()
        local sorted = {}
        for name, data in pairs(Settings.AutomationPriorities) do
            if data.Selected then
                table.insert(sorted, { name = name, priority = data.Priority })
            end
        end
        table.sort(sorted, function(a, b) return a.priority < b.priority end)
        return sorted
    end

    local cachedSorted = BuildSortedPriority()
    local lastPriorityRebuild = 0
    local priorityRebuildInterval = 2 -- only re-sort if settings changed

    local cachedSelected = GetSelectedBosses()
    local lastBossSelectionTick = 0
    local bossSelectionInterval = 1

    while task.wait(0.2) do
        local now = tick()

        -- Rebuild priority order occasionally
        if now - lastPriorityRebuild >= priorityRebuildInterval then
            lastPriorityRebuild = now
            cachedSorted = BuildSortedPriority()
        end

        -- Refresh selected bosses occasionally
        if now - lastBossSelectionTick >= bossSelectionInterval then
            lastBossSelectionTick = now
            cachedSelected = GetSelectedBosses()
        end

        local resolved = nil
        for _, entry in ipairs(cachedSorted) do
            if entry.name == "Auto Bosses" and Settings.AutoBoss then
                if next(cachedSelected) then
                    local list = ScanBosses(cachedSelected)
                    if #list > 0 then
                        resolved = "boss"
                        break
                    end
                end
            elseif entry.name == "Auto Level Farm" and Settings.AutoLevel then
                resolved = "farm"
                break
            elseif entry.name == "Auto Dungeon" and Settings.AutoDungeon then
                resolved = "dungeon"
                break
            elseif entry.name == "Auto Specific Mobs" and Settings.SpecificFarmEnabled then
                resolved = "specific"
                break
            elseif entry.name == "Auto Closest Mobs" and Settings.ClosestFarmEnabled then
                resolved = "closest"
                break
            end
        end

        ActiveAutomation = resolved
    end
end)

local function GetBossTeleportCFrame(bossHrp)
    local dist = Settings.AutoBossPosition and 7 or Settings.TpDistance
    local targetPos = bossHrp.Position
    if Settings.TpStyle == "Above" then
        return CFrame.new(targetPos + Vector3.new(0, dist, 0), targetPos)
    elseif Settings.TpStyle == "Below" then
        return CFrame.new(targetPos + Vector3.new(0, -dist, 0), targetPos)
    elseif Settings.TpStyle == "Behind" then
        return CFrame.new(targetPos + (bossHrp.CFrame.LookVector * -dist), targetPos)
    elseif Settings.TpStyle == "Front" then
        return CFrame.new(targetPos + (bossHrp.CFrame.LookVector * dist), targetPos)
    end
    return CFrame.new(targetPos + Vector3.new(0, dist, 0), targetPos)
end

task.spawn(function()
    local lastStatusTick  = 0
    local lastEquipTick   = 0
    local lastNextScanTick = 0
    local STATUS_INTERVAL  = 0.15
    local EQUIP_INTERVAL   = 2.0
    local NEXT_SCAN_INTERVAL = 1.0

    while true do
        -- ── Idle check ───────────────────────────────────────
        if not Settings.AutoBoss then
            if Settings.BossCurrentTarget then
                Settings.BossCurrentTarget = nil
                StopMovement()
                BossStatusParagraph:SetText(BossStatusText(false))
            end
            task.wait(0.3)
            continue
        end

        -- ── Player alive check ───────────────────────────────
        local pChar = LocalPlayer.Character
        local pHum  = pChar and pChar:FindFirstChild("Humanoid")
        if not pChar or not pHum or pHum.Health <= 0 then
            Settings.BossCurrentTarget = nil
            task.wait(1)
            continue
        end

        -- ── Boss selection check ─────────────────────────────
        local selectedDisplay = GetSelectedBosses()
        if not next(selectedDisplay) then
            BossStatusParagraph:SetText(BossStatusText(true, nil, 0, 0, nil, nil, Settings.AutoBossPosition, nil))
            task.wait(0.5)
            continue
        end

        -- ── Scan for alive boss ──────────────────────────────
        local bossList = ScanBosses(selectedDisplay)
        if #bossList == 0 then
            Settings.BossCurrentTarget = nil
            BossStatusParagraph:SetText(BossStatusText(true, nil, 0, 0, nil, nil, Settings.AutoBossPosition, nil))
            task.wait(0.5)
            continue
        end

        -- ── Acquire target — use exact same instance throughout ──
        local entry         = bossList[1]
        local targetModel   = entry.model
        local targetDisplay = entry.displayName
        local nextDisplay   = bossList[2] and bossList[2].displayName or nil

        -- Validate model and humanoid exist
        local humanoid = targetModel:FindFirstChild("Humanoid")
        if not targetModel.Parent or not humanoid or humanoid.Health <= 0 then
            task.wait(0.3)
            continue
        end

        -- HRP may not be loaded yet (boss is far away or still initializing).
        -- Step 1: teleport to model center so the game streams in the boss parts.
        local hrp = targetModel:FindFirstChild("HumanoidRootPart")
        if not hrp then
            local ok, cf = pcall(function() return targetModel:GetModelCFrame() end)
            if ok and cf then
                local pChar0 = LocalPlayer.Character
                local pHrp0  = pChar0 and pChar0:FindFirstChild("HumanoidRootPart")
                if pHrp0 then pHrp0.CFrame = cf end
            end
            -- Wait up to 3 seconds for HRP to stream in
            local waited = 0
            repeat
                task.wait(0.1)
                waited = waited + 0.1
                hrp = targetModel:FindFirstChild("HumanoidRootPart")
            until hrp or waited >= 3 or not targetModel.Parent
        end

        -- If HRP still missing after streaming wait, skip and retry
        if not hrp or not targetModel.Parent or humanoid.Health <= 0 then
            task.wait(0.3)
            continue
        end

        Settings.BossCurrentTarget = targetModel
        local engageStart = tick()
        lastEquipTick     = 0
        lastStatusTick    = 0
        lastNextScanTick  = 0

        HandleAutoEquip()

        if Settings.BossGateFarming then
            local islandName = GetClosestPortalIsland(hrp.Position)
            if islandName then
                BossStatusParagraph:SetText(BossStatusText(true, targetDisplay, humanoid.Health, humanoid.MaxHealth, nextDisplay, nil, Settings.AutoBossPosition, 0))
                pcall(function()
                    TeleportToPortalRemote:FireServer(islandName)
                end)
                task.wait(0.5)
            end
            if hrp and hrp.Parent then
                TeleportTo(GetBossTeleportCFrame(hrp))
            end
        end

        -- ── Engagement loop ──────────────────────────────────
        repeat
            -- Toggle check
            if not Settings.AutoBoss then
                StopMovement()
                break
            end

            -- Player death check
            pChar = LocalPlayer.Character
            pHum  = pChar and pChar:FindFirstChild("Humanoid")
            if not pChar or not pHum or pHum.Health <= 0 then
                Settings.BossCurrentTarget = nil
                StopMovement()
                local newChar = LocalPlayer.CharacterAdded:Wait()
                newChar:WaitForChild("Humanoid", 10)
                newChar:WaitForChild("HumanoidRootPart", 10)
                task.wait(1.5)
                HandleAutoEquip()
                break
            end

            -- Boss still alive? (use cached refs — only check .Parent and .Health)
            if not targetModel.Parent or humanoid.Health <= 0 then
                break
            end

            -- Periodic re-equip
            local now = tick()
            if now - lastEquipTick >= EQUIP_INTERVAL then
                lastEquipTick = now
                HandleAutoEquip()
            end

            -- Teleport (respects RoundTeleport + AutoBossPosition distance)
            TeleportTo(GetBossTeleportCFrame(hrp))

            -- Throttled status update
            if now - lastStatusTick >= STATUS_INTERVAL then
                lastStatusTick = now
                local pHrp = pChar and pChar:FindFirstChild("HumanoidRootPart")
                local dist = pHrp and (pHrp.Position - hrp.Position).Magnitude or nil
                BossStatusParagraph:SetText(BossStatusText(
                    true, targetDisplay,
                    humanoid.Health, humanoid.MaxHealth,
                    nextDisplay, dist,
                    Settings.AutoBossPosition,
                    math.floor(now - engageStart)
                ))
            end

            -- Throttled next-boss scan
            if now - lastNextScanTick >= NEXT_SCAN_INTERVAL then
                lastNextScanTick = now
                local freshList = ScanBosses(GetSelectedBosses())
                nextDisplay = nil
                for _, e in ipairs(freshList) do
                    if e.model ~= targetModel then
                        nextDisplay = e.displayName
                        break
                    end
                end
            end

            task.wait()
        until not Settings.AutoBoss
            or not targetModel.Parent
            or humanoid.Health <= 0

        -- ── Cleanup after engagement ─────────────────────────
        StopMovement()
        Settings.BossCurrentTarget = nil

        if not Settings.AutoBoss then
            BossStatusParagraph:SetText(BossStatusText(false))
        else
            task.wait(0.3) -- brief pause before scanning for next boss
        end
    end
end)

    local keyLink = "https://imphub.vercel.app/GetKeyAccess"

    if getgenv().setclipboard then
        setclipboard(keyLink)
    end

    return keyLink
end

local function ValidateKey(userKey)
    if not userKey or userKey == "" then
        return false, "Empty key provided"
    end

    local status = api.check_key(userKey)

    if status.code == "KEY_VALID" then
        script_key = userKey
        return true, "Key validation successful"
    elseif status.code == "KEY_HWID_LOCKED" then
        return false, "Key linked to different HWID"
    elseif status.code == "KEY_INCORRECT" then
        return false, "Key is wrong or deleted"
    else
        return false, status.message or "Unknown error"
    end
end

Progress.Login:SetKeyLink(GetKeyLink())

local window = Progress.Window

local function CheckKey(key)
    isLoggingIn = true
    task.spawn(window.SetLoadingState, window, isLoggingIn)

    local isValid, message = ValidateKey(key)

    isLoggingIn = false
    task.spawn(window.SetLoadingState, window, isLoggingIn)

    return isValid, message
end

Progress.Login.OnLogin = function(key)
    if not isLoggingIn then
        local keyResult, message = CheckKey(key)

        if keyResult then
            task.spawn(
                function()
                    Progress:ShowToast(
                        {
                            Text = "Valid Key!",
                            Duration = 2.5
                        }
                    )
                    TheScript()
                end
            )

            if writefile then
                if isfile("ImpKey.txt") and delfile then
                    delfile("ImpKey.txt")
                end
                
                writefile("ImpKey.txt", key)
                
                
                task.spawn(function()
                    Progress:ShowToast(
                        {
                            Text = "Saved Key Successfully!",
                            Duration = 4.5
                        }
                    )
                end)
            end
        else
            task.spawn(
                window.ShowDialog,
                window,
                {
                    Title = "Error Checking Key",
                    Text = "Your key may be expired or invalid due to an HWID mismatch. If you recently switched or reinstalled executors, please reset your hwid. You can try validating again to make sure it's a key error. This can also happen if the executor doesn't support our key system. Error: " ..
                        (message or "Unknown error")
                }
            )
        end
    end
end
